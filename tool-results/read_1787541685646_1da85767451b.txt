   507→        function handleDisplayMethodChange() {
   508→            const isFull = document.getElementById('dFull').checked;
   509→            const termsInput = document.getElementById('termsCount');
   510→            let currentVal = parseFloat(termsInput.value);
   511→            
   512→            if (isFull) {

   518→                }

   524→        function updateVal(id, step) {
   525→            const el = document.getElementById(id);
   526→            let val = parseFloat(el.value);
   527→            val = Math.round((val + step) * 10) / 10; 
   528→            if (val < 0.1) val = 0.1; 
   529→            
   530→            // Smart Limits
   531→            if (id === 'termsCount') { 
   532→                if(val < 2) val = 2; 
   533→                if(val > maxTerms) val = maxTerms; 
   534→            }
   535→            if (id.includes('Time')) { if(val > 10) val=10; }
   536→            
   537→            el.value = val;
   538→            
   539→            // Update labels
   540→            if(id === 'termsCount') document.getElementById('termsCountValue').innerText = val;
   541→            if(id === 'displayTime') document.getElementById('displayTimeValue').innerText = val + 's';
   542→            if(id === 'disappearTime') document.getElementById('disappearTimeValue').innerText = val + 's';
