   437→   437→        function generateQuestion() {
   438→   438→            const getNum = (digits) => {
   439→   439→                if (digits === 1) return Math.floor(Math.random() * 9) + 2; // Divisor > 1
   440→   440→                const min = Math.pow(10, digits - 1);
   441→   441→                const max = Math.pow(10, digits) - 1;
   442→   442→                return Math.floor(Math.random() * (max - min + 1)) + min;
   443→   443→            };
   444→   444→            
   445→   445→            let dividend, divisor, quotient;
   446→   446→            
   447→   447→            while(true) {
   448→   448→                divisor = getNum(gameSettings.divisorLength);
   449→   449→                // To ensure the dividend is large enough, let's derive it
   450→   450→                const minQuotient = 2;
   451→   451→                const maxQuotient = Math.pow(10, gameSettings.dividendLength - gameSettings.divisorLength) -1;
   452→   452→                const safeMaxQuotient = Math.max(minQuotient, maxQuotient);
   453→   453→                
   454→   454→                quotient = Math.floor(Math.random() * (safeMaxQuotient - minQuotient + 1)) + minQuotient;
   455→   455→                
   456→   456→                dividend = divisor * quotient;
   457→   457→
   458→   458→                // Check if dividend respects the number of digits
   459→   459→                const dividendStr = dividend.toString();
   460→   460→                if (dividendStr.length <= gameSettings.dividendLength) {
   461→   461→                    break;
   462→   462→                }
   463→   463→            }
   464→   464→            
   465→   465→            return { terms: [dividend, '÷', divisor], answer: quotient };
   466→   466→        }
