   391→        function generateQuestion() {
   392→            let { termsCount, numberLength } = gameSettings;
   393→            const max = Math.pow(10, numberLength);
   394→            let currentTotal = Math.floor(Math.random() * max);
   395→            const terms = [currentTotal];
   396→            for(let i=1; i<termsCount; i++) {
   397→                let op = Math.random() > 0.5 ? '+' : '-';
   398→                let num = Math.floor(Math.random() * max);
   399→                if(op==='+' && currentTotal+num < 10000000) { terms.push('+', num); currentTotal+=num; }
   400→                else if(op==='-' && currentTotal-num >= 0) { terms.push('-', num); currentTotal-=num; }
   401→                else { terms.push('+', 0); }
   402→            }
   403→            let qStr = terms[0]; for(let i=1; i<terms.length; i+=2) qStr += ` ${terms[i]} ${terms[i+1]}`;
   404→            return { text: qStr, answer: currentTotal, terms: terms };
   405→        }

   417→            const box = document.getElementById('questionBox');
   418→            let html = `<div class="vertical-math animate__animated animate__fadeIn">`;
   419→            html += `<div class="math-row">${terms[0]}</div>`;
   420→            for (let i = 1; i < terms.length; i+=2) {
   421→                const op = terms[i];
   422→                const num = terms[i+1];
   423→                const isSub = (op === '-');
   424→                const styleClass = isSub ? 'text-red-500' : ''; 
   425→                const sign = isSub ? '<span class="op-sign">-</span>' : '';

   427→            }

   430→            setTimeout(fitText, 0); setTimeout(fitText, 50);

   436→        function displayQuestionSequentially(terms) {
   437→            const box = document.getElementById('questionBox');
   438→            box.innerHTML = '';
   439→            let i = 0;
   440→            function showNext() {
   441→                if (!isGameRunning) return;
   442→                if (i < terms.length) {
   443→                    let val = (i > 0) ? (terms[i] === '+' ? terms[i+1] : `-${terms[i+1]}`) : terms[i];
   444→                    let colorClass = (i > 0 && terms[i] === '-') ? 'text-red-500' : 'text-white';
   445→                    box.innerHTML = `<div class="seq-num text-7xl font-bold animate__animated animate__zoomIn ${colorClass}" style="font-family:'Chakra Petch'">${val}</div>`;
   446→                    fitText(); 
   447→                    termDisplayTimeout = setTimeout(() => {
   448→                        if (!isGameRunning) return;
   449→                        box.innerHTML = '';
   450→                        setTimeout(showNext, gameSettings.disappearTime);
   451→                    }, gameSettings.displayTime);
   452→                    i += (i === 0 ? 1 : 2);
   453→                } else {
   454→                    box.innerHTML = `<div class="text-6xl text-yellow-400 animate-bounce">؟</div>`;
   455→                    isAnswering = true;
   456→                    questionStartTime = Date.now();
   457→                    startQuestionTimer();
   458→                }
   459→            }
   460→            showNext();
   461→        }

   513→        async function handlePdfAndExit() {
   514→            Swal.fire({ title: 'يتم تحميل ملف ال PDF...', html: 'يرجى الانتظار قليلاً...', allowOutsideClick: false, didOpen: () => { Swal.showLoading(); } });
