   106→function sendGmailSMTP($to, $subject, $body) {
   107→    $host = 'ssl://' . SMTP_HOST;
   108→    $port = SMTP_PORT;
   109→    $username = SMTP_USER;
   110→    $password = str_replace(' ', '', SMTP_PASS);
   111→
   112→    $socket = fsockopen($host, $port, $errno, $errstr, 15);
   113→    if (!$socket) return "خطأ اتصال: $errstr ($errno)";
   114→
   115→    fgets($socket, 515);
   116→
   117→    function checkCmd($socket, $cmd, $expected) {
   118→        fputs($socket, $cmd . "\r\n");
   119→        $response = "";
   120→        while($str = fgets($socket, 515)) {
   121→            $response .= $str;
   122→            if(substr($str, 3, 1) == " ") break;
   123→        }
   124→        if (substr($response, 0, 3) != $expected) return $response;
   125→        return true;
   126→    }
   127→
   128→    if (($res = checkCmd($socket, "EHLO " . $_SERVER['SERVER_NAME'], 250)) !== true) return "EHLO Error: $res";
   129→    if (($res = checkCmd($socket, "AUTH LOGIN", 334)) !== true) return "AUTH Error: $res";
   130→    if (($res = checkCmd($socket, base64_encode($username), 334)) !== true) return "User Error: $res";
   131→    if (($res = checkCmd($socket, base64_encode($password), 235)) !== true) return "فشل المصادقة: $res";
   132→    if (($res = checkCmd($socket, "MAIL FROM: <$username>", 250)) !== true) return "FROM Error: $res";
   133→    if (($res = checkCmd($socket, "RCPT TO: <$to>", 250)) !== true) return "TO Error: $res";
   134→    if (($res = checkCmd($socket, "DATA", 354)) !== true) return "DATA Error: $res";
   135→
   136→    $headers  = "MIME-Version: 1.0\r\n";
   137→    $headers .= "Content-type: text/html; charset=UTF-8\r\n";
   138→    $headers .= "From: ساحة العباقرة <$username>\r\n";
   139→    $headers .= "To: $to\r\n";
   140→    $headers .= "Subject: =?UTF-8?B?".base64_encode($subject)."?=\r\n";
   141→
   142→    fputs($socket, "$headers\r\n$body\r\n.\r\n");
   143→    
   144→    $response = "";
   145→    while($str = fgets($socket, 515)) {
   146→        $response .= $str;
   147→        if(substr($str, 3, 1) == " ") break;
   148→    }
   149→    
   150→    if (substr($response, 0, 3) != 250) return "Body Error: $response";
   151→
   152→    fputs($socket, "QUIT\r\n");
   153→    fclose($socket);
   154→    return true;
   155→}
