   304→function sendGmailSMTP($to, $subject, $body) {
   305→    $host = 'ssl://' . SMTP_HOST;
   306→    $port = SMTP_PORT;
   307→    $username = SMTP_USER;
   308→    $password = str_replace(' ', '', SMTP_PASS);
   309→
   310→    $socket = fsockopen($host, $port, $errno, $errstr, 10);
   311→    if (!$socket) return false;
   312→
   313→    fgets($socket, 515);
   314→
   315→    function checkCmd($socket, $cmd, $expected) {
   316→        fputs($socket, $cmd . "\r\n");
   317→        $response = "";
   318→        while($str = fgets($socket, 515)) {
   319→            $response .= $str;
   320→            if(substr($str, 3, 1) == " ") break;
   321→        }
   322→        if (substr($response, 0, 3) != $expected) return false;
   323→        return true;
   324→    }
   325→
   326→    if (!checkCmd($socket, "EHLO " . $_SERVER['SERVER_NAME'], 250)) return false;
   327→    if (!checkCmd($socket, "AUTH LOGIN", 334)) return false;
   328→    if (!checkCmd($socket, base64_encode($username), 334)) return false;
   329→    if (!checkCmd($socket, base64_encode($password), 235)) return false;
   330→    if (!checkCmd($socket, "MAIL FROM: <$username>", 250)) return false;
   331→    if (!checkCmd($socket, "RCPT TO: <$to>", 250)) return false;
   332→    if (!checkCmd($socket, "DATA", 354)) return false;
   333→
   334→    $headers  = "MIME-Version: 1.0\r\n";
   335→    $headers .= "Content-type: text/html; charset=UTF-8\r\n";
   336→    $headers .= "From: ساحة العباقرة <$username>\r\n";
   337→    $headers .= "To: $to\r\n";
   338→    $headers .= "Subject: =?UTF-8?B?".base64_encode($subject)."?=\r\n";
   339→
   340→    fputs($socket, "$headers\r\n$body\r\n.\r\n");
   341→    
   342→    $response = "";
   343→    while($str = fgets($socket, 515)) {
   344→        $response .= $str;
   345→        if(substr($str, 3, 1) == " ") break;
   346→    }
   347→    
   348→    if (substr($response, 0, 3) != 250) return false;
   349→
   350→    fputs($socket, "QUIT\r\n");
   351→    fclose($socket);
   352→    return true;
   353→}
