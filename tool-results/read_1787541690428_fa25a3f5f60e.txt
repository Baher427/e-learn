   437→        function generateQuestion() {
   438→            const getNum = (digits) => {
   439→                if (digits === 1) return Math.floor(Math.random() * 9) + 2; // Divisor > 1
   440→                const min = Math.pow(10, digits - 1);
   441→                const max = Math.pow(10, digits) - 1;
   442→                return Math.floor(Math.random() * (max - min + 1)) + min;
   443→            };
   444→            
   445→            let dividend, divisor, quotient;
   446→            
   447→            while(true) {
   448→                divisor = getNum(gameSettings.divisorLength);
   449→                // To ensure the dividend is large enough, let's derive it
   450→                const minQuotient = 2;
   451→                const maxQuotient = Math.pow(10, gameSettings.dividendLength - gameSettings.divisorLength) -1;
   452→                const safeMaxQuotient = Math.max(minQuotient, maxQuotient);
   453→                
   454→                quotient = Math.floor(Math.random() * (safeMaxQuotient - minQuotient + 1)) + minQuotient;
   455→                
   456→                dividend = divisor * quotient;
   457→
   458→                // Check if dividend respects the number of digits
   459→                const dividendStr = dividend.toString();
   460→                if (dividendStr.length <= gameSettings.dividendLength) {
   461→                    break;
   462→                }
   463→            }
   464→            
   465→            return { terms: [dividend, '÷', divisor], answer: quotient };
   466→        }

   468→        function nextQuestion() {
   469→            if (questionCount >= totalQuestions) {
   470→                endGame(true);
   471→                return;
   472→            }
   473→            
   474→            questionCount++;
   475→            currentQuestionNumEl.textContent = questionCount;
   476→            const progress = (questionCount / totalQuestions) * 100;
   477→            gameProgress.style.width = `${progress}%`;
   478→
   479→            currentQuestion = generateQuestion();
   480→
   481→            answerInput.value = '';
   482→            answerInput.disabled = true;
   483→            feedback.textContent = '';
   484→            
   485→            displayQuestionSequentially(currentQuestion.terms);
   486→        }

   488→        function displayQuestionSequentially(terms) {
   489→            questionBox.textContent = '';
   490→            let i = 0;
   491→            
   492→            function showNextTerm() {
   493→                if (!isGameRunning) return;
   494→
   495→                if (i < terms.length) {
   496→                    let termToShow = terms[i];
   497→                    
   498→                    questionBox.textContent = termToShow;
   499→                    questionBox.classList.add('fade-in');
   500→
   501→                    termDisplayTimeout = setTimeout(() => {
   502→                        if (!isGameRunning) return;
   503→                        questionBox.classList.remove('fade-in');
   504→
   505→                        setTimeout(() => {
   506→                            if (!isGameRunning) return;
   507→                            i++;
   508→                            showNextTerm(); 
   509→                        }, gameSettings.disappearTime);
   510→
   511→                    }, gameSettings.displayTime);
   512→
   513→                } else {
   514→                    questionBox.textContent = '؟';
   515→                    questionBox.classList.add('fade-in');
   516→                    answerInput.disabled = false;
   517→                    
   518→                    setTimeout(() => {
   519→                        answerInput.focus();
   520→                        if (gameSettings.inputMethod === 'computer') {
   521→                           answerInput.scrollIntoView({ behavior: "smooth", block: "center" });
   522→                        }
   523→                    }, 100);
   524→
   525→                    questionStartTime = Date.now();
   526→                }
   527→            }
   528→            showNextTerm();
   529→        }

   532→        function checkAnswer() {
   533→            if (answerInput.disabled) return;
   534→            const userAnswer = parseInt(answerInput.value);
   535→            const timeTaken = (Date.now() - questionStartTime) / 1000;
   536→            const isCorrect = userAnswer === currentQuestion.answer;
   537→
   538→            gameResults.push({
   539→                question: `${currentQuestion.terms[0]} ÷ ${currentQuestion.terms[2]}`,
   540→                userAnswer: isNaN(userAnswer) ? "No Answer" : userAnswer,
   541→                correctAnswer: currentQuestion.answer,
   542→                isCorrect: isCorrect,
   543→                timeTaken: timeTaken.toFixed(2)
   544→            });
   545→
   546→            if (isCorrect) {
   547→                feedback.textContent = 'رائع!';
   548→                feedback.className = 'feedback correct';
   549→            } else {
   550→                feedback.textContent = `للأسف! الإجابة الصحيحة: ${currentQuestion.answer}`;
   551→                feedback.className = 'feedback incorrect';
   552→            }
   553→            
   554→            answerInput.disabled = true;
   555→            questionBox.classList.remove('fade-in');
   556→            
   557→            if (gameSettings.inputMethod !== 'computer') {
   558→                 setTimeout(nextQuestion, 1500);
   559→            } else {
   560→                // For computer keyboard, keep focus and move to next question faster
   561→                answerInput.blur(); // Briefly blur to allow UI update
   562→                setTimeout(() => {
   563→                    nextQuestion();
   564→                }, 1500);
   565→            }
   566→        }
