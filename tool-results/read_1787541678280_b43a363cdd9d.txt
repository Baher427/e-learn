   127→function getHistoryPage($pdo, $userId) {
   128→    $page = isset($_POST['page']) ? (int)$_POST['page'] : 1;
   129→    $limit = 15;
   130→    $offset = ($page - 1) * $limit;
   131→    $stmtPvP = $pdo->prepare("SELECT m.created_at, m.winner_id, m.bet_amount, u1.student_name as p1_name, u2.student_name as p2_name, m.player1_id, m.player2_id, 'pvp' as type FROM pvp_matches m JOIN users u1 ON m.player1_id = u1.id JOIN users u2 ON m.player2_id = u2.id WHERE (m.player1_id = ? OR m.player2_id = ?) AND m.status = 'completed'");
   132→    $stmtPvP->execute([$userId, $userId]);
   133→    $pvpHistory = $stmtPvP->fetchAll(PDO::FETCH_ASSOC);
   134→    $stmtAI = $pdo->prepare("SELECT created_at, total_score as points_change, performance_notes as status, 'ai' as type FROM trainings WHERE user_id = ? AND game_type = 'AI_MATCH'");
   135→    $stmtAI->execute([$userId]);
   136→    $aiHistory = $stmtAI->fetchAll(PDO::FETCH_ASSOC);
   137→    $mergedHistory = array_merge($pvpHistory, $aiHistory);
   138→    usort($mergedHistory, function($a, $b) { return strtotime($b['created_at']) - strtotime($a['created_at']); });
   139→    $totalRecords = count($mergedHistory);
   140→    $finalHistory = array_slice($mergedHistory, $offset, $limit);
   141→    echo json_encode(['status' => 'success', 'history' => $finalHistory, 'has_more' => ($offset + $limit) < $totalRecords, 'current_page' => $page]);
   142→}

   149→function startAiGame($pdo, $userId) {

   159→    $stmt->execute([$userId]); 
   160→    $user = $stmt->fetch();

   163→    
   164→    $today = date('Y-m-d');

   169→

   171→    $genLevel = max(1, $dbLevel); 
   172→    
   173→    $qCount = 100;
   174→    $fullQuestions = MentalMathGenerator::generateBatch($genLevel, $qCount); 
   175→    $uniqueMatchId = 'AI_' . time() . '_' . mt_rand(1000, 9999);
   176→    
   177→    $botSpeed = ['min' => 2140, 'max' => 4280]; 
   178→    if ($genLevel == 2) $botSpeed = ['min' => 1800, 'max' => 3800]; 
   179→    if ($genLevel == 3) $botSpeed = ['min' => 1500, 'max' => 3500];
   180→    if ($genLevel >= 4) $botSpeed = ['min' => 1200, 'max' => 3000];

   184→    foreach ($fullQuestions as $key => $q) { if($key === 'config') continue; $answers[] = $q['a']; } 
   185→    $_SESSION['ai_game_answers_' . $uniqueMatchId] = $answers; 
   186→    $_SESSION['ai_user_answers_' . $uniqueMatchId] = []; 
   187→    $clientQuestions = []; 
   188→    foreach ($fullQuestions as $key => $q) { if($key === 'config') continue; $clientQuestions[] = ['q' => $q['q']]; } 
   189→    echo json_encode(['status' => 'success', 'match_id' => $uniqueMatchId, 'questions' => $clientQuestions, 'game_config' => $fullQuestions['config'], 'bot_config' => $botSpeed, 'level_used' => $genLevel]); 
   190→}

   211→        if (isset($userAnswers[$idx]) && $userAnswers[$idx] == $realAns) { $userCorrect++; }

   217→    $userTime = intval($_POST['user_time'] ?? 9999);

   219→    $pointsAwarded = 0;
   220→    $winReason = "";
   221→    $winnerName = "";
   222→    $resultStatus = "draw"; 
   223→    $historyStatus = "loss"; 
   224→
   225→    if ($userCorrect > $aiCorrect) {

   227→        $resultStatus = "win";
   228→        $historyStatus = "win";
   229→        $winnerName = "أنت (البطل)";
   230→        $winReason = "تفوق في دقة الإجابات";

   232→        $pointsAwarded = 50;
   233→        $resultStatus = "win";
   234→        $historyStatus = "win";
   235→        $winnerName = "أنت (البطل)";
   236→        $winReason = "تفوق في السرعة (نفس الدقة)";

   242→        $winReason = "تفوق الروبوت في الدقة";

   244→        $pointsAwarded = 0;
   245→        $resultStatus = "draw"; 
   246→        $historyStatus = "draw";

   248→             $resultStatus = "loss"; $historyStatus = "loss"; 
   249→             $winnerName = "الروبوت 🤖";
   250→             $winReason = "الروبوت كان أسرع";

   252→             $winnerName = "تعادل";

   260→        $pdo->prepare("UPDATE users SET current_status = 'idle' WHERE id = ?")->execute([$userId]);
   261→    }
   262→
   263→    logAiHistory($pdo, $userId, $pointsAwarded, $historyStatus);
   264→    unset($_SESSION[$sessionKeyAnswers]);
   265→    unset($_SESSION[$sessionKeyUser]);
   266→
   267→    echo json_encode([
   268→        'status' => 'success', 
   269→        'my_score' => $userCorrect,
   270→        'my_wrong' => $userWrong,
   271→        'opp_score' => $aiCorrect, 
   272→        'winner_name' => $winnerName,

   274→        'points_awarded' => $pointsAwarded, 
   275→        'result_status' => $resultStatus,
   276→        'user_score' => $userCorrect, 
   277→        'user_time' => $userTime,
   278→        'ai_time' => $aiTime,
   279→        'wrong' => $userWrong,

   281→        'message' => $winReason

   312→    $pdo->prepare("UPDATE pvp_matches SET status = 'completed', winner_id = ? WHERE id = ?")->execute([$winnerId, $matchId]); 
   313→    $pdo->commit(); 
   314→    
   315→    echo json_encode([
   316→        'status' => 'success', 
   317→        'message' => 'تم الانسحاب',
   318→        'my_score' => ($match['player1_id'] == $userId) ? $match['p1_score'] : $match['p2_score'],
   319→        'opp_score' => ($match['player1_id'] == $userId) ? $match['p2_score'] : $match['p1_score'],
   320→        'winner_id' => $winnerId
   321→    ]); 
   322→}

   324→function syncGameState($pdo, $userId) { 
   325→    $matchId = $_POST['match_id']; 
   326→    $stmt = $pdo->prepare("SELECT * FROM pvp_matches WHERE id = ?"); 
   327→    $stmt->execute([$matchId]); 
   328→    $match = $stmt->fetch(); 
   329→    
   330→    if (!$match) { echo json_encode(['status' => 'ended']); return; } 
   331→    
   332→    $isP1 = ($match['player1_id'] == $userId); 
   333→    
   334→    $myScore = $isP1 ? intval($match['p1_score']) : intval($match['p2_score']); 
   335→    $myProgress = $isP1 ? intval($match['p1_progress']) : intval($match['p2_progress']);
   336→    
   337→    $opponentScore = $isP1 ? intval($match['p2_score']) : intval($match['p1_score']); 
   338→    $opponentProgress = $isP1 ? intval($match['p2_progress']) : intval($match['p1_progress']);
   339→    $oppStatus = $isP1 ? $match['p2_status'] : $match['p1_status']; 
   340→    
   341→    $myWrong = max(0, $myProgress - $myScore);
   342→    $opponentWrong = max(0, $opponentProgress - $opponentScore);
   343→
   344→    if ($match['p1_status'] == 'finished' && $match['p2_status'] == 'finished' && $match['status'] != 'completed') { 
   345→        finishGameLogic($pdo, $match); 
   346→        echo json_encode(['status' => 'check_result']); 
   347→        return; 
   348→    } 
   349→    
   350→    if ($match['status'] == 'completed') { 
   351→        $winReason = "انتهاء المباراة";
   352→        $winnerName = "تعادل";
   353→        $winTitle = "تعادل 🤝";
   354→        $winMsg = "مباراة قوية! حظ أوفر في القادم.";
   355→        
   356→        $namesStmt = $pdo->prepare("SELECT id, student_name FROM users WHERE id IN (?, ?)");
   357→        $namesStmt->execute([$match['player1_id'], $match['player2_id']]);
   358→        $users = $namesStmt->fetchAll(PDO::FETCH_KEY_PAIR);
   359→        
   360→        if ($match['winner_id']) {
   361→            $winnerName = $users[$match['winner_id']];
   362→            if ($match['winner_id'] == $userId) {
   363→                 $winTitle = "🎉 مبروك يا بطل!";
   364→                 $winMsg = "فوز مستحق! تفوقت بجدارة.";
   365→            } else {
   366→                 $winTitle = "حظ أوفر 😢";
   367→                 $winMsg = "ولا يهمك، المباراة الجاية بتاعتك!";
   368→            }
   369→        }
   370→
   371→        $qData = json_decode($match['questions_json'], true); 
   372→        $config = $qData['config'] ?? ['win_points' => $match['bet_amount'], 'loss_points' => $match['bet_amount']]; 
   373→        
   374→        $result = 'draw'; 
   375→        $pointsChange = 0; 
   376→        if ($match['winner_id'] == $userId) { 
   377→            $result = 'win'; 
   378→            $pointsChange = $config['win_points']; 
   379→        } elseif ($match['winner_id'] && $match['winner_id'] != $userId) { 
   380→            $result = 'loss'; 
   381→            $pointsChange = $config['loss_points']; 
   382→        } 
   383→        
   384→        echo json_encode([
   385→            'status' => 'completed',

   503→class MentalMathGenerator {
   504→    public static function generateBatch($level, $count) {
   505→        $questions = [];
   506→        for ($i = 0; $i < $count; $i++) {
   507→            $questions[] = self::generateQuestion($level);
   508→        }

   512→    private static function generateQuestion($level) {
   513→        $numCols = 1; 
   514→        $opCols = 1;  
   515→        $minRows = 3; $maxRows = 5;
   516→
   517→        if ($level == 1) {
   518→            $numCols = 1; $opCols = 1;
   519→            $minRows = 3; $maxRows = 5;
   520→        } elseif ($level == 2) {
   521→            $numCols = 1; $opCols = 1;
   522→            $minRows = 3; $maxRows = 5;
   523→        } elseif ($level == 3) {
   524→            $numCols = 2; 
   525→            $opCols = 1;
   526→            $minRows = 3; $maxRows = 6;
   527→        } elseif ($level == 4) {
   528→            $numCols = 2;
   529→            $opCols = 2; 
   530→            $minRows = 3; $maxRows = 5;
   531→        } else {
   532→            $numCols = 2; $opCols = 2;
   533→            $minRows = 3; $maxRows = 5;
   534→        }
   535→
   536→        $floors = rand($minRows, $maxRows);
   537→        $numbers = [];
   538→        
   539→        $currentVal = array_fill(0, $numCols, 0);
   540→        for ($d = 0; $d < $opCols; $d++) {
   541→            $currentVal[$d] = rand(1, 9); 
   542→        }
   543→        $numbers[] = self::digitsToNum($currentVal);
   544→
   545→        for ($f = 1; $f < $floors; $f++) {
   546→            $validRowFound = false;
   547→            $attempts = 0;
   548→            
   549→            while (!$validRowFound && $attempts < 50) {
   550→                $attempts++;
   551→                $tempMoveDigits = array_fill(0, $opCols, 0);
   552→                $isValidRow = true;
   553→                $tempCurrentVal = $currentVal; 
   554→
   555→                for ($d = 0; $d < $opCols; $d++) {
   556→                    $allowed = self::getAllowedRules($level, $d);
   557→                    $foundDigitMove = false;
   558→                    $candidates = range(-9, 9);
   559→                    shuffle($candidates);
   560→
   561→                    foreach ($candidates as $change) {
   562→                        if ($change == 0) continue;
   563→                        $oldDigit = $tempCurrentVal[$d];
   564→                        $sum = $oldDigit + $change;
   565→                        $type = 'DIRECT';
   566→                        if ($sum >= 10 || $sum < 0) { $type = 'COMP10'; } 
   567→                        else { if ($change > 0) { if ($oldDigit < 5 && $sum >= 5) $type = 'COMP5'; } else { if ($oldDigit >= 5 && $sum < 5) $type = 'COMP5'; } }
   568→                        if (!in_array($type, $allowed)) { continue; }
   569→                        if ($type == 'COMP10') {
   570→                            $nextCol = $d + 1;
   571→                            if ($nextCol >= $numCols) { continue; }
   572→                            $carry = ($sum >= 10) ? 1 : -1;
   573→                            $nextVal = $tempCurrentVal[$nextCol] + $carry;
   574→                            if ($nextVal < 0 || $nextVal > 9) { continue; }
   575→                            $tempCurrentVal[$nextCol] = $nextVal;
   576→                            if ($sum >= 10) $sum -= 10;
   577→                            if ($sum < 0) $sum += 10;
   578→                        }
   579→                        $tempCurrentVal[$d] = $sum;
   580→                        $tempMoveDigits[$d] = $change;
   581→                        $foundDigitMove = true;
   582→                        break; 
   583→                    }
   584→
   585→                    if (!$foundDigitMove) {
   586→                        $isValidRow = false;
   587→                        break;
   588→                    }
   589→                }
   590→
   591→                if ($isValidRow) {
   592→                    $numMove = self::digitsToNum($tempMoveDigits);
   593→                    if ($numMove != 0) {
   594→                        $numbers[] = $numMove;
   595→                        $currentVal = $tempCurrentVal; 
   596→                        $validRowFound = true;
   597→                    }
   598→                }
   599→            }
   600→            
   601→            if (!$validRowFound) break;
   602→        }
   603→
   604→        $qText = ltrim(implode("\n", array_map(function($n) { return ($n > 0) ? "+$n" : "$n"; }, $numbers)), '+');
   605→        return ['q' => $qText, 'a' => array_sum($numbers)];
   606→    }

   608→    private static function digitsToNum($digitsArr) {
   609→        $num = 0;
   610→        foreach ($digitsArr as $pos => $val) {
   611→            $num += $val * pow(10, $pos);
   612→        }
   613→        return $num;
   614→    }

   616→    private static function getAllowedRules($level, $digitIndex) {
   617→        if ($level == 1) return ['DIRECT'];
   618→        if ($level == 2) return ['DIRECT', 'COMP5'];
   619→        if ($level == 3) return ['DIRECT', 'COMP5', 'COMP10']; 
   620→        if ($level == 4) { if ($digitIndex == 0) return ['DIRECT', 'COMP5']; return ['DIRECT']; }
   621→        if ($level == 5) { return ['DIRECT', 'COMP5']; }
   622→        return ['DIRECT', 'COMP5', 'COMP10'];
   623→    }
